   112→export function AdminPanel() {
   113→  const [active, setActive] = useState(false);
   114→  const [state, setState] = useState<HashState>("loading");
   115→  const [admin, setAdmin] = useState<AdminInfo | null>(null);
   116→
   117→  const checkSession = useCallback(async () => {
   118→    try {
   119→      const res = await fetch("/api/admin/me", { cache: "no-store" });
   120→      if (res.ok) {
   121→        const data = await res.json();
   122→        setAdmin(data.admin);
   123→        setState("dashboard");
   124→      } else {
   125→        setAdmin(null);
   126→        setState("auth");
   127→      }
   128→    } catch {
   129→      setAdmin(null);
   130→      setState("auth");
   131→    }
   132→  }, []);
   133→
   134→  // Activate the overlay whenever hash === "#admin".
   135→  useEffect(() => {
   136→    const onHash = () => {
   137→      const hash = window.location.hash.toLowerCase();
   138→      const isAdmin = hash === "#admin" || hash.startsWith("#admin");
   139→      setActive(isAdmin);
   140→      if (isAdmin) {
   141→        // Lock body scroll while the admin overlay is open.
   142→        document.documentElement.style.overflow = "hidden";
   143→        void checkSession();
   144→      } else {
   145→        document.documentElement.style.overflow = "";
   146→      }
   147→    };
   148→    onHash();
   149→    window.addEventListener("hashchange", onHash);
   150→    return () => {
   151→      window.removeEventListener("hashchange", onHash);
   152→      document.documentElement.style.overflow = "";
   153→    };
   154→  }, [checkSession]);
   155→
   156→  if (!active) return null;
   157→
   158→  return (
   159→    <AnimatePresence>
   160→      <motion.div
   161→        key="admin-overlay"
   162→        initial={{ opacity: 0 }}
   163→        animate={{ opacity: 1 }}
   164→        exit={{ opacity: 0 }}
   165→        transition={{ duration: 0.18 }}
   166→        className="fixed inset-0 z-[200] bg-white text-foreground"
   167→      >
   168→        {/* Brand bar accent */}
   169→        <div className="h-1.5 w-full bg-brand-red" />
   170→
   171→        <div className="h-[calc(100vh-1.5rem)] overflow-y-auto scroll-brand">
   172→          {state === "loading" && (
   173→            <div className="grid h-full place-items-center">
   174→              <div className="flex flex-col items-center gap-3 text-muted-foreground">
   175→                <Loader2 className="h-7 w-7 animate-spin text-brand-red" />
   176→                <p className="font-serif-display text-lg">Loading admin panel…</p>
   177→              </div>
   178→            </div>
   179→          )}
   180→
   181→          {state === "auth" && (
   182→            <AuthScreen
   183→              onLoggedIn={(a) => {
   184→                setAdmin(a);
   185→                setState("dashboard");
   186→              }}
   187→            />
   188→          )}
   189→
   190→          {state === "dashboard" && admin && (
   191→            <Dashboard
   192→              admin={admin}
   193→              onLogout={async () => {
   194→                await fetch("/api/admin/logout", { method: "POST" });
   195→                setAdmin(null);
   196→                setState("auth");
   197→                toast.success("Logged out.");
   198→              }}
   199→            />
   200→          )}
   201→        </div>
   202→
   203→        {/* Close button (top-right) — closes the overlay but stays on the page. */}
   204→        <button
   205→          type="button"
   206→          aria-label="Close admin panel"
   207→          onClick={() => {
   208→            // Strip the hash without leaving a # in the URL.
   209→            history.replaceState(
   210→              null,
   211→              "",
   212→              window.location.pathname + window.location.search
   213→            );
   214→            setActive(false);
   215→            document.documentElement.style.overflow = "";
   216→          }}
   217→          className="fixed right-4 top-4 z-[210] grid h-10 w-10 place-items-center rounded-full bg-brand-black/90 text-white shadow-lg hover:bg-brand-black"
   218→        >
   219→          <X className="h-5 w-5" />
   220→        </button>
   221→      </motion.div>
   222→    </AnimatePresence>
   223→  );
   224→}

   349→function LoginForm({
   350→  onLoggedIn,
   351→}: {
   352→  onLoggedIn: (a: AdminInfo) => void;
   353→}) {
   354→  const [username, setUsername] = useState("");
   355→  const [password, setPassword] = useState("");
   356→  const [busy, setBusy] = useState(false);
   357→
   358→  async function submit(e: React.FormEvent) {
   359→    e.preventDefault();
   360→    if (!username || !password) {
   361→      toast.error("Please enter your username and password.");
   362→      return;
   363→    }
   364→    setBusy(true);
   365→    try {
   366→      const res = await fetch("/api/admin/login", {
   367→        method: "POST",
   368→        headers: { "Content-Type": "application/json" },
   369→        body: JSON.stringify({ username, password }),
   370→      });
   371→      const data = await res.json();
   372→      if (!res.ok || !data.ok) {
   373→        toast.error(data.error || "Could not log in.");
   374→        return;
   375→      }
   376→      toast.success("Logged in.");
   377→      onLoggedIn(data.admin);
   378→    } catch {
   379→      toast.error("Network error. Please try again.");
   380→    } finally {
   381→      setBusy(false);
   382→    }
   383→  }
   384→
   385→  return (
   386→    <form onSubmit={submit} className="space-y-4">
   387→      <div className="space-y-1.5">
   388→        <Label htmlFor="login-username">Username</Label>
   389→        <Input
   390→          id="login-username"
   391→          value={username}
   392→          onChange={(e) => setUsername(e.target.value)}
   393→          autoComplete="username"
   394→          placeholder="admin1"
   395→          disabled={busy}
   396→        />
   397→      </div>
   398→      <div className="space-y-1.5">
   399→        <Label htmlFor="login-password">Password</Label>
   400→        <PasswordInput
   401→          id="login-password"
   402→          value={password}
   403→          onChange={(e) => setPassword(e.target.value)}
   404→          autoComplete="current-password"
   405→          placeholder="••••••••"
   406→          disabled={busy}
   407→        />
   408→      </div>
   409→      <Button
   410→        type="submit"
   411→        className="w-full bg-brand-red hover:bg-brand-red/90"
   412→        disabled={busy}
   413→      >
   414→        {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : null}
   415→        Log in
   416→      </Button>
   417→    </form>
   418→  );
   419→}

   421→function RegisterForm({

   534→function ForgotForm({ onDone }: { onDone: () => void }) {
   535→  const [username, setUsername] = useState("");
   536→  const [waLink, setWaLink] = useState<string | null>(null);
   537→  const [code, setCode] = useState("");
   538→  const [newPassword, setNewPassword] = useState("");
   539→  const [busy, setBusy] = useState(false);
   540→  const [stage, setStage] = useState<"request" | "reset">("request");
   541→
   542→  async function requestCode(e: React.FormEvent) {
   543→    e.preventDefault();
   544→    if (!username) {
   545→      toast.error("Please enter your username.");
   546→      return;
   547→    }
   548→    setBusy(true);
   549→    try {
   550→      const res = await fetch("/api/admin/forgot", {
   551→        method: "POST",
   552→        headers: { "Content-Type": "application/json" },
   553→        body: JSON.stringify({ username }),
   554→      });
   555→      const data = await res.json();
   556→      if (!res.ok || !data.ok) {
   557→        toast.error(data.error || "Could not generate reset code.");
   558→        return;
   559→      }
   560→      if (data.waLink) {
   561→        setWaLink(data.waLink);
   562→        setStage("reset");
   563→        toast.success("Reset code generated. Tap the WhatsApp button below.");
   564→      } else {
   565→        toast.info(data.message || "If that username exists, a reset has been prepared.");
   566→      }
   567→    } catch {
   568→      toast.error("Network error. Please try again.");
   569→    } finally {
   570→      setBusy(false);
   571→    }
   572→  }
   573→
   574→  async function submitReset(e: React.FormEvent) {
   575→    e.preventDefault();
   576→    if (!username || !code || !newPassword) {
   577→      toast.error("All fields are required.");
   578→      return;
   579→    }
   580→    setBusy(true);
   581→    try {
   582→      const res = await fetch("/api/admin/reset", {
   583→        method: "POST",
   584→        headers: { "Content-Type": "application/json" },
   585→        body: JSON.stringify({ username, code, newPassword }),
   586→      });
   587→      const data = await res.json();
   588→      if (!res.ok || !data.ok) {
   589→        toast.error(data.error || "Could not reset password.");
   590→        return;
   591→      }
   592→      toast.success("Password reset. You can log in now.");
   593→      onDone();
   594→    } catch {
   595→      toast.error("Network error. Please try again.");
   596→    } finally {
   597→      setBusy(false);
   598→    }
   599→  }
   600→
   601→  if (stage === "reset" && waLink) {
   602→    return (
   603→      <div className="space-y-4">
   604→        <a
   605→          href={waLink}
   606→          target="_blank"
   607→          rel="noopener noreferrer"
   608→          className="flex items-center justify-center gap-2 w-full rounded-lg bg-[#25D366] px-4 py-3 text-white font-semibold shadow hover:bg-[#1ebe57] transition"
   609→        >
   610→          <MessageCircle className="h-5 w-5" />
   611→          Open WhatsApp to receive your reset code
   612→        </a>
   613→        <p className="text-xs text-muted-foreground">
   614→          A WhatsApp chat with your own number opens pre-filled with the code.
   615→          Tap send, then read the code below. (We can&apos;t programmatically
   616→          send WhatsApp messages without the Business API, so this self-message
   617→          is the channel.)
   618→        </p>
   619→        <Separator />
   620→        <form onSubmit={submitReset} className="space-y-4">
   621→          <div className="space-y-1.5">
   622→            <Label htmlFor="reset-code">6-digit code</Label>
   623→            <Input
   624→              id="reset-code"
   625→              value={code}
   626→              onChange={(e) =>
   627→                setCode(e.target.value.replace(/[^\d]/g, "").slice(0, 6))
   628→              }
   629→              inputMode="numeric"
